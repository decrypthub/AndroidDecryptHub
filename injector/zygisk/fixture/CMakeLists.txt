cmake_minimum_required(VERSION 3.22)
project(adh_zygisk_fixture CXX)
string(LENGTH "${ADH_P5_NONCE}" nonce_length)
if(NOT ADH_P5_NONCE MATCHES "^[a-f0-9]+$" OR NOT nonce_length EQUAL 12)
  message(FATAL_ERROR "explicit 12-hex test namespace required")
endif()
configure_file(test_namespace.h.in test_namespace.h @ONLY)
add_library(adh SHARED ../jni/main.cpp)
target_compile_options(adh PRIVATE -std=c++17 -fno-exceptions -fno-rtti -Wall -Wextra -Oz -include ${CMAKE_CURRENT_BINARY_DIR}/test_namespace.h)
target_link_libraries(adh log dl)
# Compile the same source without overrides, so the default contract also gets a build signal.
add_library(adh_default_contract SHARED ../jni/main.cpp)
target_compile_options(adh_default_contract PRIVATE -std=c++17 -fno-exceptions -fno-rtti -Wall -Wextra -Oz)
target_link_libraries(adh_default_contract log dl)
